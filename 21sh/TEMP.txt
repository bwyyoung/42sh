all		: $(NAME)

$(NAME)	: $(OBJ)
	@make -C libft/
	@make -C ft_printf/
	@$(CC) $(FLAGS) $(INC) $(OBJ) -o $(NAME) $(LIB)

%.o: %.c
	@$(CC) -o $@ -c $< $(FLAGS)

clean	:
	@make -C libft/ clean
	@make -C ft_printf/ clean
	@$(RM) $(OBJ)

fclean	: clean
	@make -C libft/ fclean
	@make -C ft_printf/ fclean
	@$(RM) $(NAME)

re		: fclean all

.PHONY : re fclean clean all
